refugee_origin_prop <- unhcr_level1 |>
    mutate(region_of_origin2 = case_when(region_of_origin %in% c("Stateless", "Unknown") ~ "Other*",
        .default = region_of_origin
    )) |>
    group_by(Year, region_of_origin2) |>
    summarize(total_refugees = sum(num_refugees)) |>
    ungroup() |>
    group_by(Year) |>
    mutate(prop = total_refugees / sum(total_refugees))

plot_prop <- refugee_origin_prop |>
    ggplot(aes(x = Year, y = prop, color = region_of_origin2)) +
    geom_line() +
    scale_color_ghro("region") +
    scale_y_continuous("proportion of the world's refugees", labels = scales::percent_format(), expand = c(0, 0)) +
    labs(x = NULL) +
    theme_minimal_grid() + 
    labs(caption = "* Other includes refugees of unknown origin and those who are stateless")

    plot_prop

  1. Since 2021, the proportion of the world’s refugees from Europe has risen 20%.

  2. The proportion from Asia has dropped from a majority of about 3 in 5 refugees to about 2 in 5 refugees.

  3. For the first time, the highest proportion of refugees is from Africa, not Asia, though both are around 40%.
