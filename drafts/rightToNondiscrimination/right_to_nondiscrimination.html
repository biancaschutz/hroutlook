library(ghRo)

gender_ts <- perceptions |> filter(Sex != "BOTHSEX", Grounds.of.discrimination == "ALL", Age == "ALLAGE", Disability.status == "_T", Location == "ALLAREA", Education == "_T", Income == "_T", TimePeriod > 2015) |> 
select(TimePeriod, Sex, GeoAreaCode, Value) |>
pivot_wider(names_from = "Sex", values_from = "Value") |>
# first, we are dropping any country-year pairs that don't have data for both men and women
drop_na() |>
# then, we are binning years
mutate(bin = case_when(between(TimePeriod, 2016, 2020) ~ "2016-2020", 
between(TimePeriod, 2021, 2025) ~ "2021-2025", 
.default = NA)) |>
group_by(GeoAreaCode, bin) |>
# for each country and time range, only take most most recent
filter(TimePeriod == max(TimePeriod)) |>
ungroup() |> 
pivot_longer(c(FEMALE, MALE), names_to = "Sex") |>
select(-TimePeriod) |> 
# finally, we only take countries that have data for both time bins
pivot_wider(names_from = "bin", values_from = "value") |>
drop_na() |>
pivot_longer(c(`2016-2020`, `2021-2025`)) |>
group_by(Sex, name) |>
summarize(m = mean(value)) |>
mutate(Sex = str_to_title(Sex)) |>
pivot_wider(names_from = "name", values_from = "m")

Methodology was as follows: 1. For each country, took the most recent data value from 2016-2020 and then for 2021-2025. For example, if Argentina had values for 2017 and 2019, its value for 2016-2020 is 2019. 2. Then, I narrowed it down to only countries that had data for both time periods. There were 27 countries that had data a) for both male and female for the same year, and b) for both time periods.

urls <- c("lfp", "emp", "vaw", "mar")
labels <- c("overarching", "employment", "violence", "marriage")
params <- list(
    region = TRUE,
    country = TRUE,
    year = TRUE,
    series = TRUE,
    natureEstimate = TRUE
)

gender_eq <- list()

for (i in 1:length(urls)) {
    body <- list(
        limit = 1000
    )

    response <- POST(
        url = paste0("https://hrdx-api-staging.un.org/sgLglGeneq", urls[i], "/search"),
        body = toJSON(body, auto_unbox = TRUE),
        content_type_json()
    )

    results <- fromJSON(content(response, "text"), flatten = TRUE)$results

    results$name <- labels[[i]]

    gender_eq[[labels[i]]] <- results
}

ge_regions <- gender_equality |> 
filter(year == 2024, !is.na(countryId)) |> 
left_join(ghRo::countries |> rename(country = name), by = c("countryId" = "countryId")) |>
select(name, year, level1, value) |>
mutate(level1 = as.character(level1)) |>
left_join(ghRo::regions |> select(regionId, name) |> rename(region = name), by = c("level1" = "regionId")) |>
group_by(region, name) |>
summarize(avg_unwt = mean(value), 
n = n())
