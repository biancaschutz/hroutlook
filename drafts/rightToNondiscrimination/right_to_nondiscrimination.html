urls <- c("lfp", "emp", "vaw", "mar")
labels <- c("overarching", "employment", "violence", "marriage")
params <- list(
    region = TRUE,
    country = TRUE,
    year = TRUE,
    series = TRUE,
    natureEstimate = TRUE
)

gender_eq <- list()

for (i in 1:length(urls)) {
    body <- list(
        limit = 1000
    )

    response <- POST(
        url = paste0("https://hrdx-api-staging.un.org/sgLglGeneq", urls[i], "/search"),
        body = toJSON(body, auto_unbox = TRUE),
        content_type_json()
    )

    results <- fromJSON(content(response, "text"), flatten = TRUE)$results

    results$name <- labels[[i]]

    gender_eq[[labels[i]]] <- results
}
